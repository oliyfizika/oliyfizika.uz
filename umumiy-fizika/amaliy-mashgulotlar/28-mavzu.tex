\section{Gidravlik press}



\begin{enumerate}
\item Gidravlik press katta porshenining yuzi 5 dm$^2$, 
kichik porsheniniki esa 20 cm$^2$ ga teng. 
Press kichik porshenini 500 N kuch bilan bossak, 
katta porshenida qancha kuch hosil bo‘ladi (kN)?

\begin{enumerate}[label=\Alph*)]
\item 20
\item 30
\item 12,5
\item 25
\end{enumerate}


\item Gidravlik pressning 10 cm$^2$ yuzaga ega bo‘lgan kichik porsheniga 
100 N kuch ta’sir etsa, yuzi 1 m$^2$ bo‘lgan katta porshenida 
qanday kuch paydo bo‘ladi (N)?

\begin{enumerate}[label=\Alph*)]
\item 100000
\item 10000
\item 1000
\item 100
\end{enumerate}


\item Gidravlik press katta porshenining yuzi 375 cm$^2$. 
Uning kichik porsheniga 160 N kuch bilan ta’sir qilib, 
og‘irligi 12 kN bo‘lgan yukni ko‘tarish uchun 
kichik porshenning yuzi qanday (cm$^2$) bo‘lishi kerak?

\begin{enumerate}[label=\Alph*)]
\item 5
\item 8
\item 4
\item 2,5
\end{enumerate}


\item Gidravlik domkrat kichik porshenining yuzi 2,4 cm$^2$ bo‘lib, 
unga ta’sir qiluvchi kuch 1000 N. 
Katta porshen yuzi 2880 cm$^2$ bo‘lsa, bu domkrat 
necha tonna massali yukni ko‘tarishi mumkin?

\begin{enumerate}[label=\Alph*)]
\item 5
\item 10
\item 20
\item 100
\end{enumerate}


\item Rasmda tasvirlangan sistemada katta porshen siljimasligi uchun 
uning ustiga qanday yuk qo‘yilishi lozim (kg)? 
$S_2=4,5S_1$.

\begin{enumerate}[label=\Alph*)]
\item 2,5
\item 5
\item 10
\item 22,5
\end{enumerate}


\item Gidravlik press kichik porshenining yuzi 8 cm$^2$, 
katta porsheniniki 800 cm$^2$. 
Kichik porshenga 600 N kuch berilganda, katta porshendan 
54 kN kuch olindi. Ishqalanish bo‘lmaganda va ishqalanish bo‘lganda, 
shu press yordamida kuchni necha marta oshirish mumkin bo‘ladi?

\begin{enumerate}[label=\Alph*)]
\item 67,5; 75
\item 90; 100
\item 100; 90
\item 75; 67,5
\end{enumerate}


\item Gidravlik press kichik porshenining yuzi 4 cm$^2$, 
kattasiniki esa 500 cm$^2$. 
Agar kichik porshenni 25 cm/s tezlik bilan tushirsak, 
katta porshen qanday tezlik bilan ko‘tariladi (cm/s)?

\begin{enumerate}[label=\Alph*)]
\item 1
\item 0,6
\item 0,4
\item 0,2
\end{enumerate}


\item Gidravlik pressning kichik porsheni 60 N kuch ta’siri ostida 
15 cm pastga tushdi. Bunda katta porshen 5 cm yuqoriga ko‘tarilsa, 
katta porshenga necha Nyuton kuch ta’sir qilgan?

\begin{enumerate}[label=\Alph*)]
\item 600
\item 180
\item 220
\item 100
\end{enumerate}


\item Gidravlik press kichik porsheniga yelkalari 1:5 nisbatda bo‘lgan 
richag yordamida ta’sir ko‘rsatiladi. 
Agar gidravlik press porshenlarining yuzalari 4 marta farq qilsa, 
ushbu mexanizm kuchdan umumiy qancha yutuq beradi?

\begin{enumerate}[label=\Alph*)]
\item 25
\item 15
\item 10
\item 20
\end{enumerate}

\end{enumerate}
